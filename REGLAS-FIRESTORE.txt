==========================================================
   🔑 REGLAS DE FIRESTORE — sikodark radio (versión reparada)
==========================================================
Cómo aplicarlas:
1. https://console.firebase.google.com → proyecto sikodarkradio-26a07
2. Firestore Database → Reglas
3. Borrar todo y pegar esto → Publicar

Cambios importantes vs. la versión anterior:
• temas_pedidos: la COLA PÚBLICA de la web necesita LECTURA anónima
  (si no, la cola muestra "permiso denegado"). La creación queda
  validada; update/delete solo el admin (el puente con service
  account bypassa estas reglas, no necesita permiso).
• config: lectura anónima, escritura solo admin; los votos de la encuesta
  siguen permitidos para cualquier oyente autenticado.
• Fallback final: todo lo demás denegado por defecto.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function logueado() { return request.auth != null; }
    function admin()    { return request.auth.token.email == "sikoradio@hotmail.com"; }

    // === USUARIOS ===
    match /usuarios/{docId} {
      allow read: if logueado();
      allow create, update: if logueado();
      allow delete: if admin();
    }

    // === PRESENCIA (latido de oyentes) ===
    match /presencia/{docId} {
      allow read, write: if logueado();
    }

    // === VISITAS Y ESCUCHAS (estadísticas) ===
    match /visitas/{docId} {
      allow create: if logueado();
      allow read: if logueado();
    }
    match /escuchas/{docId} {
      allow create: if logueado();
      allow read: if admin();
    }

    // === ME GUSTAS ===
    match /megustas/{docId} {
      allow read: if logueado();
      allow create: if logueado();
      allow delete: if admin();
    }

    // === TOKENS DE NOTIFICACIÓN PUSH ===
    match /tokens/{docId} {
      allow read, write: if logueado();
    }

    // === CONFIG (avisos, publicidad, encuesta) ===
    match /config/{docId} {
      allow read: if logueado();
      allow write: if admin();
      // Los votos de la encuesta los suma el cliente con arrayUnion
      allow update: if logueado()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['encuesta']);
    }

    // === PEDIDOS DE TEMAS ===
    // Lectura anónima: la cola en vivo de la web la ve todo oyente.
    // Creación validada: campos obligatorios y largo máximo.
    // Update/delete: solo admin. El puente del DJ usa service account
    // (Admin SDK) y por eso ignora estas reglas.
    match /temas_pedidos/{docId} {
      allow read: if logueado();
      allow create: if logueado()
        && request.resource.data.banda is string
        && request.resource.data.banda.size() <= 60
        && request.resource.data.tema is string
        && request.resource.data.tema.size() <= 60
        && request.resource.data.fecha is int;
      allow update, delete: if admin();
    }

    // Todo lo demás, bloqueado por defecto
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

==========================================================
   🧪 PRUEBAS
==========================================================
A) Web (celular, anónimo):
   1. Abrir la app → "Programá tu tema" → enviar banda + tema.
   2. Debe aparecer "✅ Programado" y el pedido en la cola en vivo
      con estado ⏳ Pendiente (ahora la cola SÍ puede leer).

B) Panel admin (PC):
   1. Abrir avisos.html → login con sikoradio@hotmail.com.
   2. En "🎸 Pedidos de temas" debe listarse el pedido; los botones
      Borrar y "Limpiar todos" funcionan (admin).

C) Errores comunes:
   • "Missing or insufficient permissions" → reglas no publicadas aún;
     esperar 1-2 min y recargar.
   • "auth/operation-not-allowed" → Authentication → Sign-in method →
     habilitar Anónimo.
   • El puente del DJ (puente_radioboss.py) marca en_cola/sonado sin
     tocar estas reglas porque usa service account.
