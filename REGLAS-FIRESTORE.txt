Por qué esta versión unifica los dos archivos anteriores:
• El puente puente2-radioboss.js entra con auth anónima (REST) y
  necesita poder marcar estado (en_cola / sonado / no_encontrado)
  en temas_pedidos: por eso se permite update SOLO sobre los
  campos de seguimiento, nunca sobre banda/tema/nombre.
• El admin (sikoradio@hotmail.com) tiene control total: borrar
  pedidos, escribir config, gestionar usuarios.
• El script limpieza_cola.py usa la cuenta de servicio (Admin SDK)
  y por eso IGNORA estas reglas: borra igual aunque acá no
  figure el permiso de borrado anónimo.
• Los votos de la encuesta los suma el cliente con arrayUnion sobre
  el campo "encuesta" (esa update puntual sí se permite).

BLOQUE PARA PEGAR EN FIREBASE CONSOLE
----------------------------------------------------------

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function admin() {
      return request.auth != null
        && request.auth.token.email == "sikoradio@hotmail.com";
    }

    // ===== COLA DE PEDIDOS (web pública + puente del DJ) =====
    match /temas_pedidos/{docId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
        && request.resource.data.estado == "pendiente"
        && request.resource.data.fecha is int
        && request.resource.data.banda is string
        && request.resource.data.banda.size() <= 60
        && (!("tema" in request.resource.data)
            || (request.resource.data.tema is string
                && request.resource.data.tema.size() <= 60));
      // update del puente anónimo: solo campos de estado/seguimiento
      allow update: if admin()
        || (request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(['estado', 'mensaje', 'archivo', 'insertadoAt',
                         'revisadoAt', 'sonadoAt', 'fecha_estado']));
      allow delete: if admin();
    }

    // ===== USUARIOS / PRESENCIA =====
    match /usuarios/{docId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if admin();
    }
    match /presencia/{docId} {
      allow read, write: if request.auth != null;
    }

    // ===== ESTADÍSTICAS =====
    match /visitas/{docId} {
      allow create: if request.auth != null;
      allow read: if admin();
    }
    match /escuchas/{docId} {
      allow create: if request.auth != null;
      allow read: if admin();
    }
    match /megustas/{docId} {
      allow read, create: if request.auth != null;
      allow delete: if admin();
    }

    // ===== TOKENS PUSH (FCM) =====
    match /tokens/{docId} {
      allow create, update: if request.auth != null;
      allow read, delete: if admin();
    }

    // ===== CONFIG (avisos, publicidad, encuesta, pedidos on/off) =====
    match /config/{docId} {
      allow read: if request.auth != null;
      allow write: if admin();
      // votos de la encuesta: arrayUnion sobre el campo "encuesta"
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data)
             .affectedKeys().hasOnly(['encuesta']);
    }

    // ===== TODO LO DEMÁS: denegado por defecto =====
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

----------------------------------------------------------
PRUEBAS RÁPIDAS
----------------------------------------------------------
A) Web anónima:
   1. Abrir la web → "Programá tu tema" → enviar banda + tema.
   2. Debe aparecer "✅ Programado" y el pedido con ⏳ Pendiente.
   3. La cola se lee sin candados de permiso.

B) Puente del DJ (auth anónima REST):
   1. Tomar un pendiente → marcarlo en_cola (si las reglas no
      lo permiten, en la ventana negra verás HTTP 403/permission
      denied en el PATCH).

C) Admin (sikoradio@hotmail.com):
   1. panel-dj.html / avisos.html: pausar pedidos y borrar pedidos
      debe funcionar; con otra cuenta el write se rechaza.

D) Errores comunes:
   • "Missing or insufficient permissions" → reglas no publicadas;
     esperar 1-2 min y recargar.
   • "auth/operation-not-allowed" → Authentication → Sign-in method
     → habilitar Anónimo y Email/Password.
   • El borrado automático de la cola vieja lo hace limpieza_cola.py
     con cuenta de servicio (no depende de estas reglas).
